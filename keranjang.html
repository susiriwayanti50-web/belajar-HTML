// Tambah item ke keranjang function tambahKeranjang(id) { let p = produk.find(item => item.id === id); if (!p) return; let itemKeranjang = keranjang.find(item => item.id === id); if (itemKeranjang) { itemKeranjang.qty += 1; } else { keranjang.push({ id: p.id, nama: p.nama, harga: p.harga, qty: 1 }); } updateKeranjangUI(); alert('Barang berhasil ditambahkan ke keranjang!'); } // Buka/Tutup Modal Keranjang function toggleKeranjang() { const popup = document.getElementById('popupKeranjang'); const overlay = document.getElementById('overlay'); popup.classList.toggle('show'); overlay.classList.toggle('show'); } // Render Ulang Isi Keranjang function updateKeranjangUI() { let totalHarga = 0; let totalQty = 0; let html = ''; if (keranjang.length === 0) { html = '

Keranjang masih kosong

'; } else { keranjang.forEach((item, index) => { let subtotal = item.harga * item.qty; totalHarga += subtotal; totalQty += item.qty; html += `
${item.nama}
Rp${item.harga.toLocaleString('id-ID')}
${item.qty}
`; }); } // Update DOM Keranjang document.getElementById('isiKeranjang').innerHTML = html; document.getElementById('totalHarga').innerText = `Total: Rp${totalHarga.toLocaleString('id-ID')}`; document.getElementById('jumlahKeranjang').innerText = totalQty; } // Fungsi Mengurangi / Menambah Qty function ubahJumlahItem(index, delta) { keranjang[index].qty += delta; // Jika jumlah menjadi 0 atau kurang, hapus otomatis dari keranjang if (keranjang[index].qty <= 0) { keranjang.splice(index, 1); } updateKeranjangUI(); } // Fungsi Hapus Langsung Per Item function hapusDariKeranjang(index) { keranjang.splice(index, 1); updateKeranjangUI(); }
